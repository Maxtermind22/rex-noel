        const categoriesContainer = document.getElementById('categoriesContainer');

        const categoryTitle = document.getElementById('categoryTitle');
        const catEmoji = document.getElementById('catEmoji');

        // ==========================================
        // CHARGER UNE CATÉGORIE
        // ==========================================
        function loadCategory(categoryKey) {
            // 1. Afficher l'écran de chargement
            loadingScreen.classList.add('active');

            // 2. Cacher les catégories
            categoriesContainer.style.display = 'none';

            // 3. Après 2 secondes (temps de l'animation du T-Rex)
            setTimeout(() => {
                // Cacher l'écran de chargement
                loadingScreen.classList.remove('active');

                // Afficher les produits
                displayProducts(categoryKey);

                // Afficher le bouton retour
                backBtn.classList.remove('hidden');

                // Scroll en haut
                window.scrollTo({ top: 0, behavior: 'smooth' });
            }, 2000);
        }

        // ==========================================
        // AFFICHER LES PRODUITS
        // ==========================================
        function displayProducts(categoryKey) {
            const data = products[categoryKey];
            if (!data) return;

            // Mettre à jour le titre
            catEmoji.textContent = data.emoji;
            catName.textContent = data.name;

            // Générer les cartes produits
            productsGrid.innerHTML = '';
            data.items.forEach((item, index) => {
                const card = document.createElement('div');
                card.className = 'card';
                card.style.animationDelay = (index * 0.1) + 's';
                card.innerHTML = `
                    <div class="image-placeholder">${item.emoji}</div>
                    <div class="product-name">${item.name}</div>
                    <div class="product-desc">${item.desc}</div>
                    <div class="product-price">${item.price} <small>€</small></div>
                `;
                productsGrid.appendChild(card);
            });

            // Afficher la section produits
            productsContainer.classList.add('active');
        }

        // ==========================================
        // RETOUR AUX CATÉGORIES
        // ==========================================
        function showCategories() {
            // Cacher les produits

            // Cacher le bouton retour
            backBtn.classList.add('hidden');

            // Scroll en haut
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }
